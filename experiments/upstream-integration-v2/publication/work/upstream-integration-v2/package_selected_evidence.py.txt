"""Package an explicit immutable evidence selection into verified deterministic shards.

No recursive selection, framework imports, native builds, or array decoding.
The pack command requires a fresh output directory with an existing parent.
"""

import argparse
import gzip
import hashlib
import json
import os
import re
import stat
import sys
import tarfile
import zlib
from pathlib import Path, PurePosixPath

SELECTION_SCHEMA = "explicit-evidence-selection-v1"
PACKAGE_SCHEMA = "verified-evidence-shards-v1"
BLOCK = 1 << 20
DEFAULT_SHARD_BYTES = 1 << 30


def require(condition, message):
    if not condition:
        raise ValueError(message)


def identity(info):
    return {
        name: getattr(info, name)
        for name in (
            "st_dev",
            "st_ino",
            "st_mode",
            "st_nlink",
            "st_size",
            "st_mtime_ns",
            "st_ctime_ns",
        )
    }


def safe_path(value):
    require(isinstance(value, str) and value, "Member path must be nonempty text")
    require(not any(ord(c) < 32 or ord(c) == 127 for c in value), "Control character in path")
    require("\\" not in value, "Backslash paths are not allowed")
    path = PurePosixPath(value)
    require(not path.is_absolute() and path.as_posix() == value, "Noncanonical/absolute path")
    require(all(p not in ("", ".", "..") for p in path.parts), "Traversal path")
    require(
        path.parts[0] in ("work", "outputs") and len(path.parts) > 1,
        "Path must be below work/ or outputs/",
    )
    return path.parts


def canonical_directory(value):
    path = Path(value).absolute()
    require(path.resolve(strict=True) == path, "Directory must not traverse symlinks")
    require(path.is_dir(), "Expected an existing directory")
    return path


def open_member(root, relative):
    """Walk from a root descriptor without following any symlink component."""
    parts = safe_path(relative)
    directory_flags = os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW
    descriptor = os.open(root, directory_flags)
    try:
        for component in parts[:-1]:
            child = os.open(component, directory_flags, dir_fd=descriptor)
            os.close(descriptor)
            descriptor = child
        file_descriptor = os.open(
            parts[-1], os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=descriptor
        )
    finally:
        os.close(descriptor)
    try:
        require(stat.S_ISREG(os.fstat(file_descriptor).st_mode), f"Not a regular file: {relative}")
        return os.fdopen(file_descriptor, "rb")
    except BaseException:
        os.close(file_descriptor)
        raise


def member_stat(root, relative):
    with open_member(root, relative) as source:
        return identity(os.fstat(source.fileno()))


def read_plain_file(path, limit=64 << 20):
    """Read a small manifest, rejecting symlinks and changes during the read."""
    path = Path(path).absolute()
    require(path.resolve(strict=True) == path, "Manifest path traverses a symlink")
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
    with os.fdopen(fd, "rb") as source:
        before = identity(os.fstat(source.fileno()))
        require(stat.S_ISREG(before["st_mode"]), "Manifest is not a regular file")
        require(before["st_size"] <= limit, "Manifest exceeds the bounded JSON size limit")
        data = source.read(limit + 1)
        require(identity(os.fstat(source.fileno())) == before, "Manifest changed while reading")
    require(identity(path.stat(follow_symlinks=False)) == before, "Manifest path changed")
    require(len(data) == before["st_size"], "Manifest byte count differs")
    return data, before


def parse_json(data):
    def pairs(values):
        result = {}
        for key, value in values:
            require(key not in result, f"Duplicate JSON key: {key}")
            result[key] = value
        return result

    def nonfinite(value):
        raise ValueError(f"Nonfinite JSON number: {value}")

    return json.loads(data, object_pairs_hook=pairs, parse_constant=nonfinite)


def validate_member(row):
    require(
        set(row) == {"path", "bytes", "sha256", "executable_bits"},
        "Unexpected/missing member fields",
    )
    safe_path(row["path"])
    require(type(row["bytes"]) is int and row["bytes"] >= 0, "Invalid member size")
    require(
        isinstance(row["sha256"], str) and re.fullmatch(r"[0-9a-f]{64}", row["sha256"]),
        "Invalid member SHA-256",
    )
    require(
        type(row["executable_bits"]) is int
        and row["executable_bits"] >= 0
        and row["executable_bits"] & ~0o111 == 0,
        "Invalid executable bits",
    )


def validate_selection(selection):
    """Validate the complete selection contract without reading its source tree."""
    require(isinstance(selection, dict), "Selection must be a JSON object")
    require(set(selection) == {"schema", "workspace_root", "files"}, "Unexpected selection fields")
    require(selection["schema"] == SELECTION_SCHEMA, "Unrecognized selection schema")
    require(
        isinstance(selection["workspace_root"], str)
        and Path(selection["workspace_root"]).is_absolute(),
        "workspace_root must be absolute",
    )
    root = Path(selection["workspace_root"])
    require(
        root.as_posix() == selection["workspace_root"] and ".." not in root.parts,
        "workspace_root must be a canonical absolute path",
    )
    require(isinstance(selection["files"], list) and selection["files"], "Empty selection")
    seen = set()
    for row in selection["files"]:
        validate_member(row)
        require(row["path"] not in seen, f"Duplicate member: {row['path']}")
        seen.add(row["path"])
    members = sorted(selection["files"], key=lambda x: x["path"])
    return root, members


def load_selection(path):
    data, source_stat = read_plain_file(path)
    selection = parse_json(data)
    selected_root, members = validate_selection(selection)
    root = canonical_directory(selected_root)
    states = {}
    for row in members:
        state = member_stat(root, row["path"])
        require(state["st_size"] == row["bytes"], f"Selected size differs: {row['path']}")
        require(
            state["st_mode"] & 0o111 == row["executable_bits"],
            f"Selected executable bits differ: {row['path']}",
        )
        states[row["path"]] = state
    return root, members, states, data, source_stat


def tar_info(row):
    info = tarfile.TarInfo(row["path"])
    info.size = row["bytes"]
    info.mode = 0o644 | row["executable_bits"]
    info.mtime = info.uid = info.gid = 0
    info.uname = info.gname = ""
    return info


def member_tar_bytes(row):
    header = tar_info(row).tobuf(format=tarfile.PAX_FORMAT, encoding="utf-8")
    return len(header) + ((row["bytes"] + 511) // 512) * 512


def archive_tar_bytes(body_bytes):
    return ((body_bytes + 1024 + tarfile.RECORDSIZE - 1) // tarfile.RECORDSIZE) * tarfile.RECORDSIZE


def plan_shards(members, target):
    require(
        type(target) is int and target >= tarfile.RECORDSIZE,
        "Shard target must be at least 10240 bytes",
    )
    shards, current, body = [], [], 0
    for row in members:
        size = member_tar_bytes(row)
        if current and archive_tar_bytes(body + size) > target:
            shards.append((current, body))
            current, body = [], 0
        current.append(row)
        body += size
    if current:
        shards.append((current, body))
    return [
        {
            "name": f"evidence-{index:04d}.tar.gz",
            "members": rows,
            "payload_bytes": sum(r["bytes"] for r in rows),
            "tar_body_bytes": body,
            "uncompressed_tar_bytes": archive_tar_bytes(body),
            "oversized_single_member": len(rows) == 1 and archive_tar_bytes(body) > target,
        }
        for index, (rows, body) in enumerate(shards, 1)
    ]


class HashReader:
    def __init__(self, source):
        self.source, self.sha256, self.bytes = source, hashlib.sha256(), 0

    def read(self, size=-1):
        data = self.source.read(size)
        self.sha256.update(data)
        self.bytes += len(data)
        return data


class HashWriter:
    def __init__(self, target):
        self.target, self.sha256, self.bytes = target, hashlib.sha256(), 0

    def write(self, data):
        written = self.target.write(data)
        require(written == len(data), "Short archive write")
        self.sha256.update(data)
        self.bytes += len(data)
        return written

    def flush(self):
        self.target.flush()


class TarReader:
    def __init__(self, source, body_bytes, maximum_bytes):
        self.source, self.body_bytes, self.bytes = source, body_bytes, 0
        self.maximum_bytes = maximum_bytes

    def read(self, size=-1):
        remaining = self.maximum_bytes - self.bytes + 1
        data = self.source.read(remaining if size < 0 else min(size, remaining))
        require(self.bytes + len(data) <= self.maximum_bytes, "Excess uncompressed archive data")
        start = max(0, self.body_bytes - self.bytes)
        require(not any(data[start:]), "Nonzero data after the declared tar body")
        self.bytes += len(data)
        return data


def verify_archive(path, shard, expected_compressed=None):
    require(not path.is_symlink(), "Archive must not be a symlink")
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
    with os.fdopen(fd, "rb") as raw:
        before = identity(os.fstat(raw.fileno()))
        require(stat.S_ISREG(before["st_mode"]), "Archive must be regular")
        compressed = HashReader(raw)
        with gzip.GzipFile(fileobj=compressed, mode="rb") as zipped:
            stream = TarReader(zipped, shard["tar_body_bytes"], shard["uncompressed_tar_bytes"])
            count = 0
            with tarfile.open(fileobj=stream, mode="r|", encoding="utf-8") as archive:
                for actual in archive:
                    require(count < len(shard["members"]), "Unexpected extra archive member")
                    expected = shard["members"][count]
                    require(
                        actual.name == expected["path"],
                        "Missing/duplicate/reordered archive member",
                    )
                    safe_path(actual.name)
                    require(
                        actual.type == tarfile.REGTYPE and actual.size == expected["bytes"],
                        "Wrong tar member type/size",
                    )
                    require(
                        actual.mode == 0o644 | expected["executable_bits"], "Wrong tar permissions"
                    )
                    require(
                        actual.uid == actual.gid == actual.mtime == 0
                        and actual.uname == actual.gname == actual.linkname == "",
                        "Nondeterministic tar metadata",
                    )
                    require(set(actual.pax_headers) <= {"path"}, "Unexpected extended tar metadata")
                    source = archive.extractfile(actual)
                    require(source is not None, "Missing member payload")
                    reader = HashReader(source)
                    while reader.read(BLOCK):
                        pass
                    require(
                        reader.bytes == expected["bytes"]
                        and reader.sha256.hexdigest() == expected["sha256"],
                        f"Archived payload differs: {actual.name}",
                    )
                    count += 1
            while stream.read(BLOCK):
                pass
            require(
                stream.bytes == shard["uncompressed_tar_bytes"],
                "Uncompressed archive byte count differs",
            )
        require(count == len(shard["members"]), "Missing archive members")
        require(compressed.bytes == before["st_size"], "Compressed archive was not fully verified")
        require(identity(os.fstat(raw.fileno())) == before, "Archive changed during verification")
    require(
        identity(path.stat(follow_symlinks=False)) == before,
        "Archive path changed during verification",
    )
    result = {
        "bytes": compressed.bytes,
        "sha256": compressed.sha256.hexdigest(),
        "members_verified": count,
        "uncompressed_tar_bytes": stream.bytes,
    }
    if expected_compressed is not None:
        require(
            result["bytes"] == expected_compressed["bytes"]
            and result["sha256"] == expected_compressed["sha256"],
            "Final archive hash/size differs",
        )
    return result


def write_archive(root, path, shard, states):
    source_states = []
    with path.open("xb") as raw:
        compressed = HashWriter(raw)
        with gzip.GzipFile(
            filename="", fileobj=compressed, mode="wb", compresslevel=1, mtime=0
        ) as zipped:
            with tarfile.open(
                fileobj=zipped, mode="w|", format=tarfile.PAX_FORMAT, encoding="utf-8"
            ) as archive:
                for row in shard["members"]:
                    with open_member(root, row["path"]) as source:
                        before = identity(os.fstat(source.fileno()))
                        require(
                            before == states[row["path"]],
                            f"Source changed before read: {row['path']}",
                        )
                        reader = HashReader(source)
                        archive.addfile(tar_info(row), reader)
                        after = identity(os.fstat(source.fileno()))
                        require(before == after, f"Source changed during read: {row['path']}")
                        require(
                            reader.bytes == row["bytes"]
                            and reader.sha256.hexdigest() == row["sha256"],
                            f"Source payload hash differs: {row['path']}",
                        )
                    require(
                        member_stat(root, row["path"]) == before,
                        f"Source path changed after read: {row['path']}",
                    )
                    source_states.append({"path": row["path"], "before": before, "after": after})
        raw.flush()
        os.fsync(raw.fileno())
    expected = {"bytes": compressed.bytes, "sha256": compressed.sha256.hexdigest()}
    verified = verify_archive(path, shard, expected)
    return {**verified, "source_stat_checks": source_states}


def write_json(path, value):
    temporary = path.with_name(path.name + ".tmp")
    with temporary.open("x") as handle:
        json.dump(value, handle, indent=2, sort_keys=True, allow_nan=False)
        handle.write("\n")
        handle.flush()
        os.fsync(handle.fileno())
    os.replace(temporary, path)


def pack(selection_path, output, target=DEFAULT_SHARD_BYTES):
    root, members, states, raw_selection, selection_stat = load_selection(selection_path)
    output = Path(output).absolute()
    canonical_directory(output.parent)
    require(not output.exists() and not output.is_symlink(), "Output directory must be fresh")
    shards = plan_shards(members, target)
    output.mkdir()
    receipt = {
        "schema": PACKAGE_SCHEMA,
        "complete": False,
        "verified": False,
        "workspace_root": str(root),
        "selection_sha256": hashlib.sha256(raw_selection).hexdigest(),
        "selection_stat": selection_stat,
        "shard_target_uncompressed_bytes": target,
        "format": {
            "tar": "pax",
            "gzip_level": 1,
            "gzip_mtime": 0,
            "tar_mtime": 0,
            "owner_ids": 0,
            "permissions": "0644 OR original executable bits",
            "member_order": "lexical POSIX relative path",
        },
        "packager_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "python": sys.version,
        "zlib": {"build": zlib.ZLIB_VERSION, "runtime": zlib.ZLIB_RUNTIME_VERSION},
        "member_count": len(members),
        "payload_bytes": sum(r["bytes"] for r in members),
        "shards": [{**s, "status": "planned"} for s in shards],
    }
    try:
        with (output / "input-selection.json").open("xb") as handle:
            handle.write(raw_selection)
        write_json(output / "package.json", receipt)
        for shard in receipt["shards"]:
            shard["status"] = "writing"
            write_json(output / "package.json", receipt)
            partial = output / (shard["name"] + ".partial")
            result = write_archive(root, partial, shard, states)
            final = output / shard["name"]
            # Exclusive creation prevents overwriting an unexpected final name.
            os.link(partial, final, follow_symlinks=False)
            partial.unlink()
            shard.update(status="verified", verification=result)
            write_json(output / "package.json", receipt)
        for row in members:
            require(
                member_stat(root, row["path"]) == states[row["path"]],
                f"Source changed during packaging: {row['path']}",
            )
        after_selection, after_stat = read_plain_file(selection_path)
        require(
            after_selection == raw_selection and after_stat == selection_stat,
            "Selection changed during packaging",
        )
        receipt.update(complete=True, verified=True)
        write_json(output / "package.json", receipt)
        return verify_package(output)
    except BaseException as error:
        receipt.update(
            complete=False,
            verified=False,
            error={"type": type(error).__name__, "message": str(error)},
        )
        for shard in receipt["shards"]:
            if shard["status"] == "writing":
                shard["status"] = "failed"
        # Preserve partial archives and completed shards as failure evidence.
        write_json(output / "package.json", receipt)
        raise


def verify_package(output, expected_package_sha256=None):
    output = canonical_directory(output)
    raw, before = read_plain_file(output / "package.json")
    if expected_package_sha256 is not None:
        require(
            hashlib.sha256(raw).hexdigest() == expected_package_sha256,
            "Package receipt does not match the trusted external hash",
        )
    receipt = parse_json(raw)
    require(
        receipt["schema"] == PACKAGE_SCHEMA
        and receipt["complete"] is True
        and receipt["verified"] is True
        and "error" not in receipt,
        "Package is incomplete/failed",
    )
    selection_raw, _ = read_plain_file(output / "input-selection.json")
    require(
        hashlib.sha256(selection_raw).hexdigest() == receipt["selection_sha256"],
        "Selection binding differs",
    )
    selection = parse_json(selection_raw)
    selected_root, members = validate_selection(selection)
    require(
        receipt["workspace_root"] == str(selected_root),
        "Package and selection workspace roots differ",
    )
    plan = plan_shards(members, receipt["shard_target_uncompressed_bytes"])
    require(len(plan) == len(receipt["shards"]), "Wrong shard inventory")
    require(
        receipt["member_count"] == len(members)
        and receipt["payload_bytes"] == sum(r["bytes"] for r in members),
        "Wrong package inventory counts",
    )
    expected_files = {"package.json", "input-selection.json", *[s["name"] for s in plan]}
    require(
        {p.name for p in output.iterdir()} == expected_files, "Unexpected/missing package files"
    )
    verified = []
    for expected, actual in zip(plan, receipt["shards"], strict=True):
        require(
            all(actual.get(k) == value for k, value in expected.items())
            and actual["status"] == "verified",
            "Shard plan changed",
        )
        states = actual["verification"]["source_stat_checks"]
        require(
            [state["path"] for state in states] == [row["path"] for row in expected["members"]],
            "Missing/reordered source stat checks",
        )
        for state, row in zip(states, expected["members"], strict=True):
            require(
                state["before"] == state["after"]
                and state["before"]["st_size"] == row["bytes"]
                and state["before"]["st_mode"] & 0o111 == row["executable_bits"],
                "Source stat checks disagree with the selected member",
            )
        verified.append(
            {
                "name": actual["name"],
                **verify_archive(output / actual["name"], actual, actual["verification"]),
            }
        )
    after_raw, after = read_plain_file(output / "package.json")
    require(after_raw == raw and after == before, "Package receipt changed during verification")
    return {
        "verified": True,
        "package_sha256": hashlib.sha256(raw).hexdigest(),
        "selection_sha256": receipt["selection_sha256"],
        "member_count": len(members),
        "shards": verified,
    }


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    commands = parser.add_subparsers(dest="command", required=True)
    packing = commands.add_parser("pack")
    packing.add_argument("--selection", type=Path, required=True)
    packing.add_argument("--output", type=Path, required=True)
    packing.add_argument("--shard-bytes", type=int, default=DEFAULT_SHARD_BYTES)
    verify = commands.add_parser("verify")
    verify.add_argument("--package", type=Path, required=True)
    verify.add_argument("--expected-package-sha256")
    args = parser.parse_args()
    result = (
        pack(args.selection, args.output, args.shard_bytes)
        if args.command == "pack"
        else verify_package(args.package, args.expected_package_sha256)
    )
    print(json.dumps(result, indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
